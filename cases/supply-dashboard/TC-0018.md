preconditions: |
  Log in as an Admin user role
steps:
  - id: s-SVC0TPS9
    action: |
      Open the Supply Dashboard menu item
    expected: |
      The 'Supply Dashboard' page is opened
  - id: s-KYYV88X7
    action: |
      Click on the column management button
    expected: |
      The modal window with the list of columns is opened
  - id: s-N7PDEAZV
    action: |
      Drag any column to rearrange its position within the dropdown list
    expected: |
      -   Users can drag and drop a column to any position within the dropdown list
      -   While dragging a column, other columns split to indicate the drop location
      -   The dragged column adjusts its position in the table accordingly
  - id: s-9BZGWC8G
    action: |
      Drag and drop each remaining column individually to rearrange their position within the dropdown list
    expected: |
      -   Users can drag and drop each column to any position within the dropdown list
      -   While dragging a column, other columns split to indicate the drop location
      -   The dragged column adjusts its position in the table accordingly
  - id: s-8GR15Z2B
    action: |
      Pin several columns in the dropdown list
    expected: |
      -   Columns appear as pinned in the dropdown, users can pin up to 5 columns
      -   The pinned columns are moved to the top of the list
      -   The pinned columns are separated from non-pinned columns by a line
      -   Pinned columns move to the beginning of the table and stay fixed during horizontal scrolling
      -   The columns are separated from the rest of the columns in the table by a vertical border
  - id: s-2K7JEN71
    action: |
      Drag any pinned column to rearrange its position within the pinned columns section
    expected: |
      -   Users can drag and drop a pinned column to any position within the pinned columns section
      -   While dragging a column, other columns split to indicate the drop location
      -   The dragged column adjusts its position in the table accordingly
  - id: s-H7PDSNY9
    action: |
      Drag any pinned column to the unpinned columns section
    expected: |
      -   Users can not drag and drop a pinned column to the unpinned columns section
      -   Pinned columns can be reordered only among other pinned columns
  - id: s-G81BHC7H
    action: |
      Drag any unpinned column to rearrange its position within the unpinned columns section
    expected: |
      -   Users can drag and drop an unpinned column to any position within the unpinned columns section
      -   While dragging a column, other columns split to indicate the drop location
      -   The dragged column adjusts its position in the table accordingly
  - id: s-8JVGTQ0D
    action: |
      Drag any unpinned column to the pinned columns section
    expected: |
      -   Users can not drag and drop an unpinned column to the pinned columns section
      -   Unpinned columns can be reordered only among other unpinned columns
  - id: s-ETSGJEC8
    action: |
      Unpin a column from the pinned columns section
    expected: |
      -   If a pinned column is unpinned, it moves to the first position of the unpinned section while keeping other unpinned columns in their order
      -   The unpinned column adjusts its position in the table accordingly
  - id: s-WDP0Z62Z
    action: |
      Uncheck a column from the pinned columns section
    expected: |
      -   If a pinned column is unchecked, it moves to the first position of the unpinned section while keeping other unpinned columns in their order
      -   The unchecked column is hidden in the table
  - id: s-J70HJJ2H
    action: |
      -   Reload the page or navigate to a different page and then return
      -   Click on the Column Management button
    expected: |
      -   The order of pinned and unpinned columns should be maintained according to the user's selection
      -   On page reload, the table should restore the column's order
