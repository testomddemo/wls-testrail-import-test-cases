preconditions: |
  Log in as an Admin user role
steps:
  - id: s-RA3KNSY6
    action: |
      Open the Supply Dashboard menu item
    expected: |
      The 'Supply Dashboard' page is opened
  - id: s-CTSQ18SB
    action: |
      Click on the column management button
    expected: |
      The modal window with the list of columns is opened
  - id: s-F1RVR37X
    action: |
      Click on the Pin button for any column
    expected: |
      -   The column appears as pinned in the dropdown
      -   The column is moved to the top of the list
      -   Pinned column moves to the beginning of the table and stays fixed during horizontal scrolling
      -   The pinned column is separated from non-pinned columns by a line
      -   The column is separated from the rest of the columns in the table by a vertical border
  - id: s-JB708AST
    action: |
      -   Click on the Column Management button
      -   Click on the Pin button for the pinned column
    expected: |
      -   The column appears as unpinned and the pin icon is displayed in gray in the dropdown
      -   If a column is unpinned, it moves to the first position of the unpinned section while keeping other unpinned columns in their custom order
  - id: s-NP97Y8AF
    action: |
      -   Click on the Column Management button
      -   Click the Pin button for multiple columns in the dropdown
    expected: |
      -   Users can select multiple (up to 5) columns from the list to pin
      -   The columns appear as pinned in the dropdown
      -   The pinned columns are moved to the top of the list
      -   The pinned columns are separated from non-pinned columns by a line
      -   The pinned columns move to the beginning of the table and stay fixed during horizontal scrolling
      -   The columns are separated from the rest of the columns in the table by a vertical border
  - id: s-SKB29M4A
    action: |
      -   Click on the Column Management button
      -   Try to pin more than 5 columns in the dropdown
    expected: |
      -   'It's possible to pin up to five columns simultaneously.' warning appears
      -   Users can not pin more than 5 columns simultaneously
  - id: s-53T1V31D
    action: |
      -   Click on the Column Management button
      -   Uncheck the checkbox of any pinned column in the dropdown
    expected: |
      -   The column appears as hidden in the dropdown and removed from the table
      -   The column is moved back to the original position in the list
      -   The pin icon of the column is hidden
  - id: s-VPB2MSA9
    action: |
      -   Click on the Column Management button
      -   Check the checkbox to enable the column back
    expected: |
      -   The column is displayed as enabled and unpinned in the dropdown
      -   The pin icon is displayed in gray
      -   The column returns to its original position in the table
  - id: s-4AKTV7P6
    action: |
      -   Reload the page or navigate to a different page and then return
      -   Click on the Column Management button
    expected: |
      -   The order of pinned columns should be maintained according to the user's selection
      -   Previously pinned columns appear as pinned in the dropdown
      -   The columns are displayed at the top of the list
      -   The pinned columns are separated from non-pinned columns by a line
      -   The columns are displayed at the beginning of the table and stay fixed during horizontal scrolling
      -   The columns are separated from the rest of the columns in the table by a vertical border
  - id: s-2P6MJD1M
    action: |
      Unpin all the columns
    expected: |
      -   All columns appear as unpinned and the pin icons are displayed in gray
      -   If columns are unpinned, they move to the first position of the unpinned section while keeping other unpinned columns in their custom order
